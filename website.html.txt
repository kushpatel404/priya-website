        // Music functionality
        let musicPlaying = false;
        function toggleMusic() {
            const music = document.getElementById('bgMusic');
            const btn = document.getElementById('musicBtn');

                btn.textContent = '🎵';
                musicPlaying = false;

            button.style.position = 'fixed';
            button.style.left = randomX + 'px';
            button.style.top = randomY + 'px';

        }

        function sayNo() {
            noClickCount++;
            const noBtn = document.getElementById('noBtn');
            const counter = document.getElementById('noCounter');
            const noCount = document.getElementById('noCount');
            
            counter.style.display = 'block';
            noCount.textContent = noClickCount;
            counter.style.animationIterationCount = 1;
            setTimeout(() => {
                counter.style.animation = 'none';
                setTimeout(() => {
                    counter.style.animation = 'shake 0.5s';
                }, 10);
            }, 500);

            // Show cute message
            showCuteMessage(cuteMessages[Math.min(noClickCount - 1, cuteMessages.length - 1)]);
            
            if (noClickCount === 1) {
                noBtn.textContent = "Are you sure? 🥺";
            } else if (noClickCount === 2) {
                noBtn.textContent = "Please? 🥹";
            } else if (noClickCount === 3) {
                noBtn.textContent = "Pretty please? 💕";
            } else if (noClickCount >= 4) {
                noBtn.textContent = "Fine, I'll keep asking! 😤";
                // Make Yes button bigger
                const yesBtn = document.querySelector('.btn-yes');
                yesBtn.style.transform = 'scale(' + (1 + noClickCount * 0.1) + ')';
            }

            // Shake the card
            const card = document.querySelector('.card');
            card.style.animation = 'none';
            setTimeout(() => {
                card.style.animation = 'shake 0.5s';
            }, 10);
        }

        function showCuteMessage(message) {
            const messageDiv = document.getElementById('cuteMessage');
            messageDiv.textContent = message;
            messageDiv.style.display = 'block';
            
            setTimeout(() => {
                messageDiv.style.display = 'none';
            }, 2000);
        }

        // Yes button - show celebration
        function sayYes() {
            const questionContent = document.getElementById('questionContent');
            const successMessage = document.getElementById('successMessage');
            const loveLetter = document.getElementById('loveLetter');
            const countdown = document.getElementById('countdown');
            
            questionContent.style.display = 'none';
            successMessage.style.display = 'block';
            
            // Show love letter after 1 second
            setTimeout(() => {
                loveLetter.style.display = 'block';
            }, 1000);

            // Show countdown after 2 seconds
            setTimeout(() => {
                countdown.style.display = 'block';
                startCountdown();
            }, 2000);
            
            // Create confetti
            createConfetti();
            
            // Create fireworks
            createFireworks();
            
            // Play celebration
            celebrate();
        }

        function createConfetti() {
            const colors = ['#ff6b6b', '#ffd93d', '#6bcf7f', '#4d96ff', '#ff8ecf'];
            
            for (let i = 0; i < 150; i++) {
                setTimeout(() => {
                    const confetti = document.createElement('div');
                    confetti.classList.add('confetti');
                    confetti.style.left = Math.random() * 100 + '%';
                    confetti.style.top = '-10px';
                    confetti.style.background = colors[Math.floor(Math.random() * colors.length)];
                    confetti.style.width = Math.random() * 10 + 5 + 'px';
                    confetti.style.height = confetti.style.width;
                    confetti.style.animationDuration = (Math.random() * 2 + 2) + 's';
                    confetti.style.borderRadius = Math.random() > 0.5 ? '50%' : '0';
                    document.body.appendChild(confetti);
                    
                    setTimeout(() => confetti.remove(), 4000);
                }, i * 30);
            }
        }

        function createFireworks() {
            for (let i = 0; i < 5; i++) {
                setTimeout(() => {
                    const x = Math.random() * window.innerWidth;
                    const y = Math.random() * window.innerHeight * 0.5;
                    
                    for (let j = 0; j < 30; j++) {
                        const firework = document.createElement('div');
                        firework.className = 'firework';
                        firework.style.left = x + 'px';
                        firework.style.top = y + 'px';
                        firework.style.background = `hsl(${Math.random() * 360}, 100%, 50%)`;
                        
                        const angle = (Math.PI * 2 * j) / 30;
                        const velocity = 100 + Math.random() * 100;
                        firework.style.setProperty('--x', Math.cos(angle) * velocity + 'px');
                        firework.style.setProperty('--y', Math.sin(angle) * velocity + 'px');
                        
                        document.body.appendChild(firework);
                        setTimeout(() => firework.remove(), 1000);
                    }
                }, i * 400);
            }
        }

        function celebrate() {
            // Add extra hearts
            for (let i = 0; i < 30; i++) {
                setTimeout(createHeart, i * 100);
            }
        }

        // Countdown to Valentine's Day
        function startCountdown() {
            const countdownTimer = document.getElementById('countdownTimer');
            const valentinesDay = new Date('February 14, 2025 00:00:00').getTime();

            const updateCountdown = () => {
                const now = new Date().getTime();
                const distance = valentinesDay - now;

                if (distance < 0) {
                    countdownTimer.innerHTML = "It's Valentine's Day! 💕";
                    return;
                }

                const days = Math.floor(distance / (1000 * 60 * 60 * 24));
                const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
                const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
                const seconds = Math.floor((distance % (1000 * 60)) / 1000);

                countdownTimer.innerHTML = `${days}d ${hours}h ${minutes}m ${seconds}s`;
            };

            updateCountdown();
            setInterval(updateCountdown, 1000);
        }

        // Add mouse move 3D effect to card
        const card = document.querySelector('.card');
        document.addEventListener('mousemove', (e) => {
            const x = (window.innerWidth / 2 - e.clientX) / 25;
            const y = (window.innerHeight / 2 - e.clientY) / 25;
            card.style.transform = `perspective(1000px) rotateY(${x}deg) rotateX(${y}deg)`;
        });

        // Keyboard shortcuts
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' || e.key === 'y' || e.key === 'Y') {
                sayYes();
            }
        });
